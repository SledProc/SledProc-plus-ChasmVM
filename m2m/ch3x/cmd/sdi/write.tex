\newcommand{\sdi!write-sentence-start}{%
\pdfsavepos%
\sdi!write-sdii-start{Sentence}%
\if-sdi-active-sentence-start-%
\global\-sdi-active-enclosing-sentence-true%
\setcounter{sdi-enclosing-sentence-counter}{\the\c@sdi-sentence-counter}%
\fi%
\stepcounter{sdi-sentence-counter}%
\stepcounter{sdi-sentence-counter!page}%
\stepcounter{sdi-sentence-counter!paragraph}%
\sdi!write-sdi-start{Sentence}%
\if-sdi-held-enclosing-sentence-%
\immediate\write\sdii-file\expandafter{\expandafter\field-mark!held%
\the\c@sdi-enclosing-sentence-counter}%
\fi
\if-sdi-active-enclosing-sentence-%
\immediate\write\sdii-file\expandafter{\expandafter\field-mark!enclosing-sentence-id%
\the\c@sdi-enclosing-sentence-counter}%
\global\-sdi-held-enclosing-sentence-true%
\global\-sdi-active-enclosing-sentence-false%
\fi%
\write\sdi-file\expandafter{\expandafter\field-mark!sentence-id%
\the\c@sdi-sentence-counter}%
\immediate\write\sdii-file\expandafter{\expandafter\field-mark!sentence-id-new%
\the\c@sdi-sentence-counter}%
\sdi!check-write-base-line%
\write\sdi-file\expandafter{\expandafter\field-mark!page%
\thepage}%
\immediate\write\sdii-file\expandafter{\expandafter\field-mark!page%
\thepage}%
\write\sdi-file\expandafter{\expandafter\field-mark!sentence-in-page%
\the\c@sdi-sentence-counter!page}%
\immediate\write\sdii-file\expandafter{\expandafter\field-mark!sentence-in-page%
\the\c@sdi-sentence-counter!page}%
\write\sdi-file\expandafter{\expandafter\field-mark!sentence-in-paragraph%
\the\c@sdi-sentence-counter!paragraph}%
\immediate\write\sdii-file\expandafter{\expandafter\field-mark!sentence-in-paragraph%
\the\c@sdi-sentence-counter!paragraph}%
\immediate\write\sdii-file\expandafter{\expandafter\field-mark!paragraph%
\the\c@sdi-paragraph-counter}%
\write\sdi-file\expandafter{\expandafter\field-mark!paragraph%
\the\c@sdi-paragraph-counter^^J%
-x\space\space\topt\pdflastxpos^^J%
-y\space\space\topt\pdflastypos^^J%
}%
\global\-sdi-active-sentence-start-true%
\global\-sdi-active-sentence-end-false%
}



\newcommand{\sdi!write-sentence-end}{%

\the\c@sdi-sentence-counter}%
\fi

}%
