\newcommand{\@sectionLevel}[2]{%
\ifcase#1\relax
\or{\sL@section{#2}}
\or{\sL@subsection{#2}}
\else{}
\fi
}


\NewDocumentCommand{\s}{E{.}{}D||{}m}{%
\IfNoValueTF{#1}{
\@sectionLevel{#2}{#3}
}
{%else
\@sectionLevel{#1}{#3}
}
}
